
(function () {
  const canvas = document.getElementById("mini-trust-canvas");
  if (!canvas) return;

  const ctx = canvas.getContext("2d");
  const width = 220;
  const height = 155;

  const dpr = Math.min(window.devicePixelRatio || 1, 2);

  canvas.width = width * dpr;
  canvas.height = height * dpr;
  ctx.scale(dpr, dpr);

  let seed = 71;

  function random() {
    seed = (seed * 1664525 + 1013904223) >>> 0;
    return seed / 4294967296;
  }

  const dots = Array.from({ length: 48 }, (_, i) => {
    const x = random() * 202;

    return {
      x: x,
      y: 130 - x * 0.48
         + (random() - 0.5) * 45,
      flat: 76 + (random() - 0.5) * 48,
      radius: 1.7 + random() * 1.1,
      phase: random() * Math.PI * 2,
      color: [
        "#2c7a2c",
        "#6b9b72",
        "#a9c5ad"
      ][i % 3]
    };
  });

  const reducedMotion = window.matchMedia(
    "(prefers-reduced-motion: reduce)"
  ).matches;

  function draw(time) {
    ctx.clearRect(0, 0, width, height);

    // Draw unlabeled axes, flush with left edge
    ctx.beginPath();
    ctx.strokeStyle = "#9ca3af";
    ctx.lineWidth = 1;
    ctx.globalAlpha = 0.65;
    ctx.setLineDash([]);

    ctx.moveTo(0.5, 0);
    ctx.lineTo(0.5, height - 0.5);
    ctx.lineTo(width, height - 0.5);
    ctx.stroke();

    ctx.globalAlpha = 1;

    const phase = (time % 9000) / 9000;

    let mix;

    if (phase < 0.28) {
      mix = 0;
    } else if (phase < 0.52) {
      mix = (phase - 0.28) / 0.24;
    } else if (phase < 0.78) {
      mix = 1;
    } else {
      mix = 1 - (phase - 0.78) / 0.22;
    }

    mix = mix * mix * (3 - 2 * mix);

    dots.forEach(dot => {
      const y =
        dot.y * (1 - mix) +
        dot.flat * mix;

      const drift = reducedMotion
        ? 0
        : Math.sin(time * 0.0007 + dot.phase) * 1.5;

      ctx.beginPath();
      ctx.arc(
        dot.x,
        y + drift,
        dot.radius,
        0,
        Math.PI * 2
      );

      ctx.fillStyle = dot.color;
      ctx.globalAlpha = 0.78;
      ctx.fill();
    });

    ctx.globalAlpha = 0.4;
    ctx.strokeStyle = "#2c7a2c";
    ctx.lineWidth = 1.3;
    ctx.setLineDash([3, 5]);

    ctx.beginPath();
    ctx.moveTo(
      0,
      123 * (1 - mix) + 76 * mix
    );
    ctx.lineTo(
      203,
      26 * (1 - mix) + 76 * mix
    );
    ctx.stroke();

    ctx.setLineDash([]);
    ctx.globalAlpha = 1;
  }

  if (reducedMotion) {
    draw(0);
  } else {
    const start = performance.now();

    function animate(now) {
      draw(now - start);
      requestAnimationFrame(animate);
    }

    requestAnimationFrame(animate);
  }
})();
